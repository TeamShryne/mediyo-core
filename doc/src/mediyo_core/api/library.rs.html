8/// Like status for library items.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub enum LikeStatus {
11    Like,
12    Dislike,
13    Indifferent,
14}
15
16fn like_endpoint(status: LikeStatus) -> &'static str {
17    match status {
18        LikeStatus::Like => "like/like",
19        LikeStatus::Dislike => "like/dislike",
20        LikeStatus::Indifferent => "like/removelike",
21    }
22}

83/// Rate a song (LIKE / DISLIKE / INDIFFERENT).
84pub fn rate_song(session: &Session, video_id: &str, status: LikeStatus) -> Result<serde_json::Value> {
85    let body = serde_json::json!({"target": {"videoId": video_id}});
86    session.request(like_endpoint(status), body)
87}

96/// Add a single video to a playlist (owned).
97pub fn add_to_playlist(session: &Session, playlist_id: &str, video_id: &str) -> Result<serde_json::Value> {
98    add_many_to_playlist(session, playlist_id, &[video_id])
99}
100
101/// Add multiple videos to a playlist at once.
102pub fn add_many_to_playlist(
103    session: &Session,
104    playlist_id: &str,
105    video_ids: &[&str],
106) -> Result<serde_json::Value> {
107    let actions: Vec<serde_json::Value> = video_ids
108        .iter()
109        .map(|vid| serde_json::json!({"action": "ACTION_ADD_VIDEO", "addedVideoId": vid}))
110        .collect();
111    let body = serde_json::json!({
112        "playlistId": playlist_id,
113        "actions": actions,
114    });
115    session.request("browse/edit_playlist", body)
116}

118/// Remove a video from a playlist (requires `setVideoId` from `get_playlist`).
119pub fn remove_from_playlist(
120    session: &Session,
121    playlist_id: &str,
122    set_video_id: &str,
123    video_id: &str,
124) -> Result<serde_json::Value> {
125    let body = serde_json::json!({
126        "playlistId": playlist_id,
127        "actions": [{
128            "action": "ACTION_REMOVE_VIDEO",
129            "setVideoId": set_video_id,
130            "removedVideoId": video_id
131        }],
132    });
133    session.request("browse/edit_playlist", body)
134}

136/// Create a new playlist.
137pub fn create_playlist(
138    session: &Session,
139    title: &str,
140    description: &str,
141    privacy: &str,
142) -> Result<String> {
143    let body = serde_json::json!({
144        "title": title,
145        "description": description,
146        "privacyStatus": privacy,
147    });
148    let resp = session.request("playlist/create", body)?;
149    resp.get("playlistId")
150        .and_then(|v| v.as_str())
151        .map(String::from)
152        .ok_or(crate::error::Error::MissingField("playlistId"))
153}
