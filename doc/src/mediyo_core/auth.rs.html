1use crate::error::{Error, Result};
2use sha1::{Digest, Sha1};

4/// Default origin used when deriving the SAPISIDHASH (site the cookies were
5/// minted for). The web client signs per-request; platforms that want stricter
6/// fidelity can pass their own origin.
7pub const DEFAULT_ORIGIN: &str = "https://music.youtube.com";
8
9/// Compute the classic `Authorization: SAPISIDHASH <ts>_<hash>` header value
10/// from the SAPISID cookie.
11///
12/// Algorithm (reverse-engineered from YouTube web):
13///   hash = SHA1("<ts> <SAPISID> <origin>")
14///   header = "SAPISIDHASH <ts>_<hash>"
15///
16/// TODO(verify): the 2026 web client sends a multi-cookie format
17/// (`SAPISIDHASH ... SAPISID1PHASH ... SAPISID3PHASH ...`). Confirm whether
18/// the classic single hash is still accepted before relying on it.
19pub fn sapisid_hash(sapisid: &str) -> Result<String> {
20    sapisid_hash_with_origin(sapisid, DEFAULT_ORIGIN)
21}
22
23pub fn sapisid_hash_with_origin(sapisid: &str, origin: &str) -> Result<String> {
24    let ts = std::time::SystemTime::now()
25        .duration_since(std::time::UNIX_EPOCH)
26        .map_err(|_| Error::Missing("system clock before unix epoch"))?
27        .as_secs();
28
29    let msg = format!("{ts} {sapisid} {origin}");
30    let mut hasher = Sha1::new();
31    hasher.update(msg.as_bytes());
32    let digest = hasher.finalize();
33    let hex: String = digest.iter().map(|b| format!("{b:02x}")).collect();
34
35    Ok(format!("SAPISIDHASH {ts}_{hex}"))
36}

42    #[test]
43    fn sapisid_hash_format() {
44        let h = sapisid_hash_with_origin("dummy_sapisid", "https://www.youtube.com").unwrap();
45        assert!(h.starts_with("SAPISIDHASH "));
46        let rest = h.trim_start_matches("SAPISIDHASH ").to_string();
47        let (ts, hash) = rest.split_once('_').unwrap();
48        assert!(ts.parse::<u64>().is_ok());
49        assert_eq!(hash.len(), 40, "sha1 hex is 40 chars");
50    }
