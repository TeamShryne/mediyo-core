1//! Text extraction: innertube text nodes are `{"runs": [...]}` or `{"simpleText": "..."}`.
2
3use serde_json::Value;
4
5/// Returns the `runs` array of a text node.
6pub fn runs(v: &Value) -> Option<&Vec<Value>> {
7    v.as_object()?.get("runs")?.as_array()
8}
9
10/// Concatenated text of all runs (or `simpleText`). Returns `None` when the
11/// node holds no text.
12pub fn text(v: &Value) -> Option<String> {
13    if let Some(tt) = v.get("simpleText").and_then(Value::as_str) {
14        return Some(tt.to_string());
15    }
16    let runs = runs(v)?;
17    let mut out = String::new();
18    for run in runs {
19        if let Some(t) = run.get("text").and_then(Value::as_str) {
20            out.push_str(t);
21        }
22    }
23    Some(out)
24}
25
26/// Iterate runs of a text node. Each run carries `text` and optionally a
27/// `navigationEndpoint` (used to link artist/album names).
28pub fn run_items(v: &Value) -> Vec<(&str, Option<&Value>)> {
29    let Some(runs) = runs(v) else {
30        return vec![];
31    };
32    runs.iter()
33        .filter_map(|r| {
34            let t = r.get("text")?.as_str()?;
35            Some((t, r.get("navigationEndpoint")))
36        })
37        .collect()
38}

40#[cfg(test)]
41mod tests {

44    use super::*;
45
46    #[test]
47    fn simple_text() {
48        assert_eq!(
49            text(&json!({ "simpleText": "hello" })),
50            Some("hello".to_string())
51        );
52    }
53
54    #[test]
55    fn runs_concat() {
56        let v = json!({ "runs": [
57            { "text": "Album" },
58            { "text": " • " },
59            { "text": "Drake" }
60        ]});
61        assert_eq!(text(&v), Some("Album • Drake".to_string()));
62    }
