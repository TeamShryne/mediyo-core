143/// Parse an album `browse` response (twoColumn layout, musicResponsiveHeaderRenderer).
144pub fn parse_album_page(resp: &Value) -> Result<AlbumPage> {
145    let header = responsive_header(resp).ok_or(Error::MissingField("album header"))?;
146    let info = parse_responsive_header(header)?;
147
148    let mut tracks = Vec::new();
149    let mut carousels = Vec::new();
150    let mut continuation = None;
151    for section in secondary_sections(resp) {
152        let Some((name, payload)) = parser::renderer(section) else {
153            continue;
154        };
155        match name {
156            "musicShelfRenderer" | "musicPlaylistShelfRenderer" => {
157                continuation = append_list_items(payload, &mut tracks)?;
158            }
159            "musicCarouselShelfRenderer" => {
160                carousels.push(parse_carousel(payload)?);
161            }
162            _ => {}
163        }
164    }
165
166    Ok(AlbumPage {
167        title: info.title,
168        artist: info.strapline,
169        year: info.year,
170        track_count: info.track_count,
171        total_duration: info.total_duration,
172        description: info.description,
173        thumbnails: info.thumbnails,
174        playlist_id: info.playlist_id,
175        tracks,
176        carousels,
177        continuation,
178    })
179}

211/// Parse a `browse` continuation response (playlist/album shelves) into the
212/// next page of items. Handles the current
213/// `onResponseReceivedActions[].appendContinuationItemsAction` shape plus the
214/// legacy `continuationContents.musicPlaylistShelfContinuation` /
215/// `sectionListContinuation` shapes.
216pub fn parse_list_continuation(resp: &Value) -> Result<ListPage> {
217    let mut items = Vec::new();
218    let mut continuation = None;
219
220    if let Some(actions) = resp
221        .get("onResponseReceivedActions")
222        .and_then(Value::as_array)
223    {
224        for action in actions {
225            if let Some(items_arr) =
226                action.pointer("/appendContinuationItemsAction/continuationItems")
227            {
228                if let Some(tok) = append_continuation_items(items_arr, &mut items)? {
229                    continuation = Some(tok);
230                }
231            }
232        }
233    }
234
235    if items.is_empty() && continuation.is_none() {
236        if let Some(shelf) = resp.pointer("/continuationContents/musicPlaylistShelfContinuation") {
237            let items_arr = shelf.get("contents").unwrap_or(&Value::Null);
238            if let Some(tok) = append_continuation_items(items_arr, &mut items)? {
239                continuation = Some(tok);
240            }
241        } else if let Some(sections) = resp
242            .pointer("/continuationContents/sectionListContinuation/contents")
243            .and_then(Value::as_array)
244        {
245            let (s, tok) = parse_sections(sections)?;
246            items.extend(s);
247            continuation = tok;
248        }
249    }
250
251    Ok(ListPage {
252        items,
253        continuation,
254    })
255}

258pub fn parse_home_continuation(resp: &Value) -> Result<HomePage> {
259    let sections = resp
260        .pointer("/continuationContents/sectionListContinuation/contents")
261        .and_then(Value::as_array)
262        .ok_or(Error::MissingField("home continuation sections"))?;
263
264    let mut carousels = Vec::new();
265    for section in sections {
266        let Some((name, payload)) = parser::renderer(section) else {
267            continue;
268        };
269        if name == "musicCarouselShelfRenderer" {
270            carousels.push(parse_carousel(payload)?);
271        }
272    }
273
274    let continuation = resp
275        .pointer("/continuationContents/sectionListContinuation/continuations")
276        .and_then(Value::as_array)
277        .and_then(|c| c.first())
278        .and_then(next_cont_token)
279        .map(String::from);
280
281    Ok(HomePage {
282        carousels,
283        continuation,
284    })

313/// Extract all items (+ trailing continuation token) from a list of
314/// `sectionListRenderer` section objects (shelves, grids, carousels).
315fn parse_sections(sections: &[Value]) -> Result<(Vec<SearchResult>, Option<String>)> {
316    let mut items = Vec::new();
317    let mut continuation = None;
318    for section in sections {
319        let Some((name, payload)) = parser::renderer(section) else {
320            continue;
321        };
322        match name {
323            "musicShelfRenderer" => {
324                let contents = payload.get("contents").unwrap_or(&Value::Null);
325                if let Some(tok) = append_continuation_items(contents, &mut items)? {
326                    continuation = Some(tok);
327                }
328            }
329            "gridRenderer" => {
330                let contents = payload.get("items").unwrap_or(&Value::Null);
331                if let Some(tok) = append_continuation_items(contents, &mut items)? {
332                    continuation = Some(tok);
333                }
334            }
335            "musicCarouselShelfRenderer" => {
336                let contents = payload.get("contents").unwrap_or(&Value::Null);
337                if let Some(tok) = append_continuation_items(contents, &mut items)? {
338                    continuation = Some(tok);
339                }
340            }
341            _ => {}
342        }
343    }
344    Ok((items, continuation))
345}

347/// Parse an artist `browse` response (singleColumn layout, musicImmersiveHeaderRenderer).
348pub fn parse_artist_page(resp: &Value) -> Result<ArtistPage> {
349    let header = resp
350        .pointer("/header/musicImmersiveHeaderRenderer")
351        .ok_or(Error::MissingField("artist header"))?;
352
353    let name = header.get("title").and_then(runs::text).unwrap_or_default();
354    let subscriber_count = header
355        .pointer("/subscriptionButton/subscribeButtonRenderer/subscriberCountText")
356        .and_then(runs::text);
357    let monthly_audience = header.get("monthlyListenerCount").and_then(runs::text);
358    let description = header.get("description").and_then(runs::text);
359    let thumbnails = thumbnails::thumbnails(header);
360
361    let mut top_songs = Vec::new();
362    let mut carousels = Vec::new();
363    let mut continuation = None;
364    let mut fallback_description = None;
365
366    let sections = resp
367        .pointer(
368            "/contents/singleColumnBrowseResultsRenderer/tabs/0/tabRenderer/content/sectionListRenderer/contents",
369        )
370        .and_then(Value::as_array);
371    if let Some(sections) = sections {
372        for section in sections {
373            let Some((name, payload)) = parser::renderer(section) else {
374                continue;
375            };
376            match name {
377                "musicShelfRenderer" => {
378                    continuation = append_list_items(payload, &mut top_songs)?;
379                }
380                "musicCarouselShelfRenderer" => carousels.push(parse_carousel(payload)?),
381                "musicDescriptionShelfRenderer" => {
382                    fallback_description = payload.get("description").and_then(runs::text);
383                }
384                _ => {}
385            }
386        }
387    }
388
389    let play_button = parse_watch_endpoint(
390        header.pointer("/playButton/buttonRenderer/navigationEndpoint"),
391    );
392    let radio_button = parse_watch_endpoint(
393        header.pointer("/startRadioButton/buttonRenderer/navigationEndpoint"),
394    );
395    let share_entity = header
396        .pointer("/shareEndpoint/shareEntityEndpoint/serializedShareEntity")
397        .and_then(Value::as_str)
398        .map(String::from);
399
400    Ok(ArtistPage {
401        name,
402        subscriber_count,
403        monthly_audience,
404        description: description.or(fallback_description),
405        thumbnails,
406        top_songs,
407        carousels,
408        continuation,
409        play_button,
410        radio_button,
411        share_entity,
412    })
413}

453/// Parse the explore page (`FEmusic_explore`) response.
454///
455/// Returns navigation buttons (New releases, Charts, Moods & genres, Podcasts)
456/// plus carousels (New albums & singles, Trending, New music videos, etc.).
457pub fn parse_explore_page(resp: &Value) -> Result<ExplorePage> {
458    let sections = resp
459        .pointer(
460            "/contents/singleColumnBrowseResultsRenderer/tabs/0/tabRenderer/content/sectionListRenderer/contents",
461        )
462        .and_then(Value::as_array)
463        .ok_or(Error::MissingField("explore sections"))?;
464
465    let mut nav_buttons = Vec::new();
466    let mut carousels = Vec::new();
467
468    for section in sections {
469        // Grid of quick-nav buttons
470        if let Some(grid) = section.get("gridRenderer") {
471            if let Some(items) = grid.get("items").and_then(Value::as_array) {
472                for item in items {
473                    if let Some(btn) = item.get("musicNavigationButtonRenderer") {
474                        let label = btn
475                            .pointer("/buttonText/runs/0/text")
476                            .and_then(Value::as_str)
477                            .unwrap_or_default()
478                            .to_string();
479                        let browse_id = btn
480                            .pointer("/clickCommand/browseEndpoint/browseId")
481                            .and_then(Value::as_str)
482                            .unwrap_or_default()
483                            .to_string();
484                        let params = btn
485                            .pointer("/clickCommand/browseEndpoint/params")
486                            .and_then(Value::as_str)
487                            .map(String::from);
488                        let icon = btn
489                            .pointer("/iconStyle/icon/iconType")
490                            .and_then(Value::as_str)
491                            .map(String::from);
492                        if !browse_id.is_empty() {
493                            nav_buttons.push(NavButton {
494                                label,
495                                browse_id,
496                                params,
497                                icon,
498                            });
499                        }
500                    }
501                }
502            }
503        }
504
505        // Carousels (same as home page)
506        if let Some((name, payload)) = parser::renderer(section) {
507            if name == "musicCarouselShelfRenderer" {
508                carousels.push(parse_carousel(payload)?);
509            }
510        }
511    }
512
513    Ok(ExplorePage {
514        nav_buttons,
515        carousels,
516    })
517}

534fn parse_responsive_header(header: &Value) -> Result<HeaderInfo> {
535    let title = header.get("title").and_then(runs::text).unwrap_or_default();
536
537    let mut year = None;
538    if let Some(runs_arr) = header
539        .get("subtitle")
540        .and_then(|s| s.get("runs"))
541        .and_then(Value::as_array)
542    {
543        for run in runs_arr {
544            if let Some(t) = run.get("text").and_then(Value::as_str) {
545                if is_year(t) {
546                    year = Some(t.to_string());
547                }
548            }
549        }
550    }
551
552    let mut strapline = None;
553    if let Some(st) = header.get("straplineTextOne") {
554        let name = runs::text(st).unwrap_or_default();
555        let id = st
556            .pointer("/runs/0/navigationEndpoint/browseEndpoint/browseId")
557            .and_then(Value::as_str);
558        if !name.is_empty() {
559            strapline = Some(ArtistRef {
560                name,
561                id: id.map(String::from),
562            });
563        }
564    }
565
566    let second_subtitle = header
567        .get("secondSubtitle")
568        .and_then(runs::text)
569        .unwrap_or_default();
570    let (track_count, total_duration) = split_track_stats(&second_subtitle);
571
572    let description = header
573        .pointer("/description/musicDescriptionShelfRenderer/description")
574        .and_then(runs::text);
575
576    let thumbnails = thumbnails::thumbnails(header);
577    let playlist_id = header
578        .pointer("/buttons")
579        .and_then(Value::as_array)
580        .and_then(|b| {
581            b.iter().find_map(|b| {
582                b.pointer(
583                    "/musicPlayButtonRenderer/playNavigationEndpoint/watchPlaylistEndpoint/playlistId",
584                )
585                .and_then(Value::as_str)
586                .or_else(|| {
587                    b.pointer("/musicPlayButtonRenderer/playNavigationEndpoint/watchEndpoint/playlistId")
588                        .and_then(Value::as_str)
589                })
590            })
591        })
592        .map(String::from);
593
594    Ok(HeaderInfo {
595        title,
596        year,
597        strapline,
598        track_count,
599        total_duration,
600        description,
601        thumbnails,
602        playlist_id,
603    })
604}

606/// Append `musicResponsiveListItemRenderer` items from a shelf, skipping the
607/// trailing `continuationItemRenderer` sentinel and returning its token.
608fn append_list_items(payload: &Value, out: &mut Vec<SearchResult>) -> Result<Option<String>> {
609    append_continuation_items(payload.get("contents").unwrap_or(&Value::Null), out)
610}
611
612/// Append list items from a `continuationItems` array (shelf, grid, or
613/// continuation response). Handles list and two-row items and returns the
614/// `continuationCommand` token if a sentinel is present.
615fn append_continuation_items(items: &Value, out: &mut Vec<SearchResult>) -> Result<Option<String>> {
616    let mut continuation = None;
617    if let Some(items) = items.as_array() {
618        for item in items {
619            let Some((rname, payload)) = parser::renderer(item) else {
620                continue;
621            };
622            match rname {
623                "musicResponsiveListItemRenderer" => out.push(parse_search_result(item)?),
624                "musicTwoRowItemRenderer" => out.push(parse_two_row_item(item)?),
625                "continuationItemRenderer" => {
626                    continuation = payload
627                        .pointer("/continuationEndpoint/continuationCommand/token")
628                        .and_then(Value::as_str)
629                        .map(String::from);
630                }
631                _ => {}
632            }
633        }
634    }
635    Ok(continuation)
636}

638fn parse_carousel(payload: &Value) -> Result<Carousel> {
639    let title = payload
640        .pointer("/header/musicCarouselShelfBasicHeaderRenderer/title")
641        .and_then(runs::text)
642        .unwrap_or_default();
643
644    let mut items = Vec::new();
645    let mut continuation = None;
646    if let Some(contents) = payload.get("contents").and_then(Value::as_array) {
647        for item in contents {
648            let Some((rname, _)) = parser::renderer(item) else {
649                continue;
650            };
651            match rname {
652                "musicTwoRowItemRenderer" => items.push(parse_two_row_item(item)?),
653                "musicNavigationButtonRenderer" => {
654                    if let Some(btn) = item.get("musicNavigationButtonRenderer") {
655                        let title = btn
656                            .get("buttonText")
657                            .and_then(runs::text)
658                            .unwrap_or_default();
659                        let browse_id = btn
660                            .pointer("/clickCommand/browseEndpoint/browseId")
661                            .and_then(Value::as_str)
662                            .map(String::from);
663                        items.push(SearchResult {
664                            category: Category::Unknown,
665                            title,
666                            artists: Vec::new(),
667                            album: None,
668                            video_id: None,
669                            browse_id,
670                            playlist_id: None,
671                            year: None,
672                            info: None,
673                            track_number: None,
674                            duration: None,
675                            thumbnails: Vec::new(),
676                            explicit: false,
677                        });
678                    }
679                }
680                "musicResponsiveListItemRenderer" => items.push(parse_search_result(item)?),
681                "musicMultiRowListItemRenderer" => {
682                    items.push(parse_multi_row_item(item)?);
683                }
684                "continuationItemRenderer" => {
685                    continuation = item
686                        .pointer("/continuationItemRenderer/continuationEndpoint/continuationCommand/token")
687                        .and_then(Value::as_str)
688                        .map(String::from);
689                }
690                _ => {}
691            }
692        }
693    }
694
695    let view_all = payload
696        .pointer(
697            "/header/musicCarouselShelfBasicHeaderRenderer/moreContentButton/buttonRenderer/navigationEndpoint/browseEndpoint",
698        )
699        .map(|ep| {
700            let browse_id = ep
701                .get("browseId")
702                .and_then(Value::as_str)
703                .unwrap_or_default()
704                .to_string();
705            let params = ep
706                .get("params")
707                .and_then(Value::as_str)
708                .map(String::from);
709            let page_type = ep
710                .pointer(
711                    "/browseEndpointContextSupportedConfigs/browseEndpointContextMusicConfig/pageType",
712                )
713                .and_then(Value::as_str)
714                .map(String::from);
715            ViewAll {
716                browse_id,
717                params,
718                page_type,
719            }
720        })
721        .filter(|v| !v.browse_id.is_empty());
722
723    Ok(Carousel {
724        title,
725        items,
726        view_all,
727        continuation,
728    })
729}

731/// Extract a continuation token from any of the known wrapper objects
732/// (`nextContinuationData`, `reloadContinuationData`, `nextRadioContinuationData`).
733/// Parse a `watchEndpoint` or `watchPlaylistEndpoint` into a [`WatchEndpoint`].
734fn parse_watch_endpoint(ep: Option<&Value>) -> Option<WatchEndpoint> {
735    let ep = ep?;
736    // Try watchEndpoint first (has videoId + playlistId)
737    if let Some(we) = ep.get("watchEndpoint") {
738        let video_id = we.get("videoId").and_then(Value::as_str).map(String::from);
739        let playlist_id = we.get("playlistId")?.as_str()?.to_string();
740        let params = we.get("params").and_then(Value::as_str).map(String::from);
741        return Some(WatchEndpoint {
742            video_id,
743            playlist_id,
744            params,
745        });
746    }
747    // Fall back to watchPlaylistEndpoint (playlistId + params, no videoId)
748    if let Some(wpe) = ep.get("watchPlaylistEndpoint") {
749        let playlist_id = wpe.get("playlistId")?.as_str()?.to_string();
750        let params = wpe.get("params").and_then(Value::as_str).map(String::from);
751        return Some(WatchEndpoint {
752            video_id: None,
753            playlist_id,
754            params,
755        });
756    }
757    None
758}

760fn next_cont_token(v: &Value) -> Option<&str> {
761    let obj = v.as_object()?;
762    for key in [
763        "nextContinuationData",
764        "reloadContinuationData",
765        "nextRadioContinuationData",
766    ] {
767        if let Some(tok) = obj
768            .get(key)
769            .and_then(|d| d.get("continuation"))
770            .and_then(Value::as_str)
771        {
772            return Some(tok);
773        }
774    }
775    None
776}

778/// Split "1.1M views • 831 tracks • 51+ hours" into (count, duration).
779/// Skips leading metadata (views) by matching on the relevant units.
780fn split_track_stats(text: &str) -> (Option<String>, Option<String>) {
781    let mut track_count = None;
782    let mut total_duration = None;
783    for part in text.split('•').map(str::trim).filter(|s| !s.is_empty()) {
784        if part.contains("song") || part.contains("track") {
785            track_count = Some(part.to_string());
786        } else if part.contains("hour") || part.contains("minute") {
787            total_duration = Some(part.to_string());
788        }
789    }
790    (track_count, total_duration)
791}
