6const BASE_URL: &str = "https://music.youtube.com/youtubei/v1";
7const USER_AGENT: &str = "Mozilla/5.0 (X11; Linux x86_64; rv:153.0) Gecko/20100101 Firefox/153.0";

9/// Cookie-based authentication (platforms supply the cookie header value).
10#[derive(Debug, Clone)]
11pub struct Auth {
12    /// Raw `Cookie` header value, e.g. `SAPISID=...; SID=...; ...`.
13    pub cookie_header: String,
14    /// SAPISID cookie value, used to derive the SAPISIDHASH Authorization header.
15    pub sapisid: Option<String>,
16}
17
18#[derive(Debug, Clone)]
19pub struct Session {
20    context: Context,
21    agent: ureq::Agent,
22    auth: Option<Auth>,
23}
24
25impl Default for Session {
26    fn default() -> Self {
27        Self::new()
28    }
29}

31impl Session {
32    /// Anonymous session (no login). Requires visitor data before most
33    /// endpoints work; see [`Session::fetch_visitor_data`].
34    pub fn new() -> Self {
35        Self {
36            context: Context::new(),
37            agent: ureq::Agent::config_builder()
38                .http_status_as_error(false)
39                .build()
40                .new_agent(),
41            auth: None,
42        }
43    }
44
45    pub fn with_context(mut self, context: Context) -> Self {
46        self.context = context;
47        self
48    }

50    /// Authenticated session built from a raw cookie header.
51    /// Pass the `SAPISID` cookie value separately so the client can compute
52    /// the SAPISIDHASH `Authorization` header.
53    pub fn with_cookies(
54        mut self,
55        cookie_header: impl Into<String>,
56        sapisid: Option<String>,
57    ) -> Self {
58        self.auth = Some(Auth {
59            cookie_header: cookie_header.into(),
60            sapisid,
61        });
62        self
63    }

65    pub fn context(&self) -> &Context {
66        &self.context
67    }

69    /// Plain HTTP GET returning the response body as text
70    /// (used for pages and player JS assets).
71    pub fn get(&self, url: &str) -> Result<String> {
72        let resp = self
73            .agent
74            .get(url)
75            .header("User-Agent", USER_AGENT)
76            .call()?;
77        Ok(resp.into_body().read_to_string()?)
78    }

80    /// POST an innertube request. `body` is merged with the session context.
81    /// The response body is decoded as JSON.
82    pub fn request(&self, endpoint: &str, body: Value) -> Result<Value> {
83        let mut body = body;
84        let context = self.context.to_json();
85        body["context"] = context["context"].clone();
86
87        let url = format!("{BASE_URL}/{endpoint}?prettyPrint=false");
88        let mut req = self
89            .agent
90            .post(&url)
91            .header("User-Agent", USER_AGENT)
92            .header("Origin", "https://music.youtube.com")
93            .header("Referer", "https://music.youtube.com/")
94            .header("X-Youtube-Client-Name", CLIENT_NAME_HEADER)
95            .header("X-Youtube-Client-Version", &self.context.client.version);
96
97        if let Some(vd) = &self.context.client.visitor_data {
98            req = req.header("X-Goog-Visitor-Id", vd);
99        }
100        if let Some(pid) = &self.context.client.page_id {
101            req = req.header("X-Goog-PageId", pid);
102        }
103        if let Some(auth) = &self.auth {
104            req = req.header("Cookie", &auth.cookie_header);
105            req = req.header("X-Youtube-Bootstrap-Logged-In", "true");
106            req = req.header("X-Goog-AuthUser", "2");
107            if let Some(sapisid) = &auth.sapisid {
108                let hash = crate::auth::sapisid_hash(sapisid)?;
109                req = req.header("Authorization", &hash);
110            }
111        }
112
113        let resp = req.send_json(body)?;
114        let status = resp.status().as_u16();
115        let text = resp.into_body().read_to_string()?;
116
117        if status >= 400 {
118            let message = extract_api_message(&text).unwrap_or_else(|| format!("HTTP {status}"));
119            return Err(Error::Api(message));
120        }
121
122        Ok(serde_json::from_str(&text)?)
123    }

125    /// Fetch a fresh `visitorData` from YouTube. The value is returned and
126    /// stored in the session context so subsequent requests carry it.
127    ///
128    /// The endpoint is a POST returning the raw visitor data string.
129    pub fn fetch_visitor_data(&mut self) -> Result<String> {
130        let url = format!("{BASE_URL}/visitor_id");
131        let body = json!({
132            "context": {
133                "client": {
134                    "clientName": "WEB",
135                    "clientVersion": "2.20240101.00.00",
136                }
137            }
138        });
139        let resp = self
140            .agent
141            .post(&url)
142            .header("Content-Type", "application/json")
143            .header("User-Agent", USER_AGENT)
144            .header("Origin", "https://music.youtube.com")
145            .header("Referer", "https://music.youtube.com/")
146            .send_json(body)?;
147        let text = resp.into_body().read_to_string()?;
148        // Current format: innertube JSON with `responseContext.visitorData`.
149        // Older format: the raw base64 string as the whole body.
150        let visitor_data = match serde_json::from_str::<Value>(&text) {
151            Ok(v) => v
152                .pointer("/responseContext/visitorData")
153                .and_then(Value::as_str)
154                .map(str::to_string)
155                .unwrap_or_else(|| text.trim().to_string()),
156            Err(_) => text.trim().to_string(),
157        };
158        self.context.client.visitor_data = Some(visitor_data.clone());
159        Ok(visitor_data)
160    }
161}

163fn extract_api_message(body: &str) -> Option<String> {
164    let v: Value = serde_json::from_str(body).ok()?;
165    v.pointer("/error/message")
166        .and_then(Value::as_str)
167        .map(String::from)
168}
